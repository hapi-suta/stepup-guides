WHAT WAS TESTED

We have a contacts table with two email columns: primary_email and secondary_email.

The business rule:
  - A record CAN have the same value in both columns (same-row duplicate allowed)
  - The same email CANNOT appear in any other record, in either column

A standard UNIQUE index cannot enforce this because it only covers one column
at a time and cannot handle the same-row exception. The solution is a
BEFORE INSERT OR UPDATE trigger that checks every write against both columns
across all other rows.

────────────────────────────────────────────────────────────────────────
STEP 1 — Create the table
────────────────────────────────────────────────────────────────────────

CREATE TABLE contacts (
    id              SERIAL PRIMARY KEY,
    name            TEXT   NOT NULL,
    primary_email   TEXT   NOT NULL,
    secondary_email TEXT
);

STEP 2 — Create the enforcement trigger
────────────────────────────────────────────────────────────────────────

CREATE OR REPLACE FUNCTION enforce_email_uniqueness()
RETURNS TRIGGER AS $$
BEGIN
    IF EXISTS (
        SELECT 1 FROM contacts
        WHERE id != COALESCE(NEW.id, 0)
          AND (primary_email   = NEW.primary_email
            OR secondary_email = NEW.primary_email)
    ) THEN
        RAISE EXCEPTION 'CONFLICT: primary_email "%" already used by another record',
            NEW.primary_email;
    END IF;

    IF NEW.secondary_email IS NOT NULL
       AND NEW.secondary_email <> NEW.primary_email THEN
        IF EXISTS (
            SELECT 1 FROM contacts
            WHERE id != COALESCE(NEW.id, 0)
              AND (primary_email   = NEW.secondary_email
                OR secondary_email = NEW.secondary_email)
        ) THEN
            RAISE EXCEPTION 'CONFLICT: secondary_email "%" already used by another record',
                NEW.secondary_email;
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_email_uniqueness
BEFORE INSERT OR UPDATE ON contacts
FOR EACH ROW EXECUTE FUNCTION enforce_email_uniqueness();
